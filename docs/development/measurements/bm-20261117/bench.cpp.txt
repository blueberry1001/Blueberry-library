#include <algorithm>
#include <chrono>
#include <cstdint>
#include <cstdlib>
#include <iostream>
#include <vector>
#include <atcoder/modint>
#include "baseline.hpp"
#include "candidate.hpp"
#include "candidate-inv.hpp"
using namespace std;
using Mint = atcoder::modint998244353;
int main(int argc, char** argv) {
  int variant = argc>1?atoi(argv[1]):0, n=argc>2?atoi(argv[2]):1000, kind=argc>3?atoi(argv[3]):0, repeat=argc>4?atoi(argv[4]):1;
  vector<Mint> a(n); uint64_t x=1729;
  for (int i=0;i<n;++i) {
    x ^= x<<7; x ^= x>>9; x ^= x<<8;
    if (kind==0) a[i] = x%998244353;
    else if (kind==1) a[i] = i>=2?a[i-1]+a[i-2]:Mint(i+1);
    else if (kind==2) a[i] = i%97==0?Mint(x%998244353):Mint(0);
    else if (kind==3) a[i] = Mint(0);
    else a[i] = i<n/2?Mint(0):Mint(x%998244353);
  }
  if (variant==2) {
    for(int m=0;m<=min(n,120);++m) { vector<Mint> b(a.begin(),a.begin()+m); if(baseline::berlekamp_massey(b)!=candidate::berlekamp_massey(b)||baseline::berlekamp_massey(b)!=candidate_inv::berlekamp_massey(b)) return 2; }
    cout<<"ok\n"; return 0;
  }
  uint64_t hash=0; auto start=chrono::steady_clock::now();
  for(int k=0;k<repeat;++k) { auto v=variant==0?baseline::berlekamp_massey(a):variant==1?candidate::berlekamp_massey(a):candidate_inv::berlekamp_massey(a); hash=hash*1000003+v.size(); for(auto y:v) hash=hash*37+y.val(); }
  cout<<chrono::duration<double,milli>(chrono::steady_clock::now()-start).count()<<' '<<hash<<'\n';
}
