using i64 = std::int64_t;
using u64 = std::uint64_t;
i64 power(i64 x, i64 e, i64 p) {
  i64 r = 1;
  for (; e; e >>= 1, x = x * x % p) if (e & 1) r = r * x % p;
  return r;
}
i64 cipolla(i64 a, i64 p) {
  a %= p; if (a < 0) a += p;
  if (a == 0 || p == 2) return a;
  if (power(a, (p - 1) / 2, p) != 1) return -1;
  i64 t = 0, w = 0;
  for (;;) {
    w = (t * t % p + p - a) % p;
    if (!w) return std::min(t, p - t);
    if (power(w, (p - 1) / 2, p) == p - 1) break;
    ++t;
  }
  using pair = std::pair<i64, i64>;
  auto product = [p, w](pair x, pair y) {
    return pair{(x.first * y.first % p + x.second * y.second % p * w) % p,
                (x.first * y.second + x.second * y.first) % p};
  };
  pair x{t, 1}, r{1, 0};
  for (i64 e = (p + 1) / 2; e; e >>= 1, x = product(x, x)) if (e & 1) r = product(r, x);
  return std::min(r.first, p - r.first);
}
bool prime(int p) { if (p < 2) return false; for (int d = 2; d * d <= p; ++d) if (p % d == 0) return false; return true; }
i64 val(int mode, i64 a, i64 p) {
  if (mode == 0) return sqrt_baseline::mod_sqrt(a, p).value_or(-1);
  if (mode == 1) return blueberry::mod_sqrt(a, p).value_or(-1);
  return cipolla(a, p);
}
int check() {
  for (int p = 2; p <= 500; ++p) if (prime(p)) {
    std::vector<int> roots(p, -1);
    for (int x = 0; x < p; ++x) { int a = x * x % p; if (roots[a] < 0) roots[a] = x; }
    for (i64 a = -2 * p; a <= 2 * p; ++a) {
      int expected = roots[(a % p + p) % p];
      for (int mode = 0; mode < 3; ++mode) if (val(mode, a, p) != expected) {
        std::cerr << "p=" << p << " a=" << a << " mode=" << mode << " expected=" << expected << " actual=" << val(mode,a,p) << '\n'; return 1;
      }
    }
  }
  std::mt19937_64 rng(20261117);
  for (i64 p : {65537LL, 998244353LL, 1000000007LL, 1000000009LL, 2013265921LL, 2147483647LL}) {
    for (int i = 0; i < 10000; ++i) {
      i64 a = i == 0 ? std::numeric_limits<i64>::min() : i == 1 ? std::numeric_limits<i64>::max() : static_cast<i64>(rng());
      i64 normalized = (a % p + p) % p, expected = val(0, a, p), actual = val(1, a, p);
      if (actual != expected || (actual >= 0 && (actual * actual % p != normalized || actual > p - actual))) {
        std::cerr << "p=" << p << " a=" << a << " expected=" << expected << " actual=" << actual << '\n'; return 1;
      }
    }
  }
  std::cout << "all checks passed\n"; return 0;
}
std::vector<std::pair<i64, i64>> make_input(int category, int count) {
  std::mt19937_64 rng(20261117);
  std::vector<std::pair<i64, i64>> result; result.reserve(count);
  const i64 ps[] = {3, 5, 7, 13, 17, 97, 257, 65537, 998244353, 1000000007, 1000000009, 2013265921, 2147483647};
  for (int i = 0; i < count; ++i) {
    i64 p = category < 3 ? 1000000007 : category < 6 ? 998244353 : category == 6 ? 2147483647 : category == 7 ? 2013265921 : ps[rng() % 13];
    i64 x = 1 + rng() % (p - 1), a = x * x % p;
    if (category == 1 || category == 4) {
      i64 non = 2; while (power(non, (p - 1) / 2, p) == 1) ++non;
      a = a * non % p;
    } else if (category == 2 || category == 5 || category >= 6) a = rng() % p;
    if ((rng() & 1) && a) a -= p;
    result.emplace_back(a, p);
  }
  return result;
}
template<class F> void timed(const std::vector<std::pair<i64, i64>>& input, int repeat, F f) {
  u64 hash = 0;
  auto start = std::chrono::steady_clock::now();
  for (int i = 0; i < repeat; ++i) for (auto [a,p] : input) hash = hash * 37 + static_cast<u64>(f(a,p) + 1);
  double ms = std::chrono::duration<double, std::milli>(std::chrono::steady_clock::now() - start).count();
  std::cout << ms << ' ' << hash;
#ifdef _WIN32
  PROCESS_MEMORY_COUNTERS info{};
  if (GetProcessMemoryInfo(GetCurrentProcess(), &info, sizeof(info))) std::cout << ' ' << info.PeakWorkingSetSize;
  else std::cout << " missing";
#endif
  std::cout << '\n';
}
int main(int argc, char** argv) {
  if (argc > 1 && std::string(argv[1]) == "check") return check();
  if (argc != 5) return 2;
  int mode = std::atoi(argv[1]), category = std::atoi(argv[2]), count = std::atoi(argv[3]), repeat = std::atoi(argv[4]);
  auto input = make_input(category, count);
  if (mode == 0) timed(input, repeat, [](i64 a, i64 p){ return sqrt_baseline::mod_sqrt(a,p).value_or(-1); });
  else if (mode == 1) timed(input, repeat, [](i64 a, i64 p){ return blueberry::mod_sqrt(a,p).value_or(-1); });
  else timed(input, repeat, [](i64 a, i64 p){ return cipolla(a,p); });
}
