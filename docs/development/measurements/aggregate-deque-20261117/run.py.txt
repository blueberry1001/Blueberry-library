import ctypes, json, os, pathlib, platform, subprocess, sys
here=pathlib.Path(__file__).resolve().parent
vc=pathlib.Path(r'C:\Program Files (x86)\Microsoft Visual Studio\2022\BuildTools\VC\Tools\MSVC\14.43.34808')
sdk=pathlib.Path(r'C:\Program Files (x86)\Windows Kits\10'); version='10.0.22621.0'
cl=vc/'bin/Hostx64/x64/cl.exe'
clang=pathlib.Path(r'C:\Program Files\Unity\Hub\Editor\6000.5.2f1\Editor\Data\PlaybackEngines\AndroidPlayer\NDK\toolchains\llvm\prebuilt\windows-x86_64\bin\clang++.exe')
env=os.environ.copy();env['PATH']=str(cl.parent)+os.pathsep+env.get('PATH','')
env['INCLUDE']=os.pathsep.join(map(str,[vc/'include',*(sdk/'Include'/version/p for p in ('ucrt','shared','um'))]))
env['LIB']=os.pathsep.join(map(str,[vc/'lib/x64',*(sdk/'Lib'/version/p/'x64' for p in ('ucrt','um'))]))
kernel=ctypes.WinDLL('kernel32',use_last_error=True);kernel.GetCurrentProcess.restype=ctypes.c_void_p
kernel.SetProcessAffinityMask.argtypes=[ctypes.c_void_p,ctypes.c_size_t]
if not kernel.SetProcessAffinityMask(kernel.GetCurrentProcess(),1<<12):raise OSError(ctypes.get_last_error())
records=[]
for compiler in ('clang','msvc'):
 for old in (True,False):
  exe=here/f'{compiler}-{old}.exe'
  defines=['-DBASELINE'] if old else []
  if compiler=='clang':cmd=[str(clang),'--target=x86_64-pc-windows-msvc','-fuse-ld=link','-std=gnu++20','-O2','-DNDEBUG',*defines,str(here/'bench.cpp'),'-o',str(exe)]
  else:cmd=[str(cl),'/nologo','/std:c++20','/EHsc','/O2','/DNDEBUG',*(['/DBASELINE'] if old else []),str(here/'bench.cpp'),'/Fo'+str(exe)+'.obj','/Fe'+str(exe)]
  cp=subprocess.run(cmd,env=env,capture_output=True,text=True,errors='replace');records.append({'compiler':compiler,'old':old,'command':cmd,'exit':cp.returncode,'output':cp.stdout+cp.stderr})
  (here/'results.json').write_text(json.dumps(records,indent=2))
  if cp.returncode:raise SystemExit(cp.returncode)
 for mode in range(5):
  for run in range(5):
   for old in ((True,False) if run%2==0 else (False,True)):
    cp=subprocess.run([str(here/f'{compiler}-{old}.exe'),str(mode)],env=env,capture_output=True,text=True,errors='replace',timeout=120)
    records.append({'compiler':compiler,'old':old,'mode':mode,'run':run,'exit':cp.returncode,'output':cp.stdout+cp.stderr})
    (here/'results.json').write_text(json.dumps(records,indent=2))
    if cp.returncode:raise SystemExit(cp.returncode)
  print(compiler,mode,flush=True)
(here/'environment.json').write_text(json.dumps({'platform':platform.platform(),'processor':platform.processor(),'cpu':12,'clang':subprocess.run([str(clang),'--version'],capture_output=True,text=True).stdout,'cl':subprocess.run([str(cl)],env=env,capture_output=True,text=True,errors='replace').stderr},indent=2))
