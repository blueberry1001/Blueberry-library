#include <chrono>
#include <cstdint>
#include <cstdlib>
#include <iostream>
#ifdef BASELINE
#include "baseline.hpp"
#else
#include "candidate.hpp"
#endif
struct F { uint64_t a,b; };
constexpr uint64_t mod=998244353;
F op(F x,F y) {return {x.a*y.a%mod,(x.b*y.a+y.b)%mod};}
F id() {return {1,0};}
uint64_t seed=42;
uint32_t rng(){seed^=seed<<7;seed^=seed>>9;return static_cast<uint32_t>(seed);}
int main(int argc,char**argv){
  int mode=std::atoi(argv[1]);
  blueberry::AggregateDeque<F,op,id> d;
  uint64_t hash=0; constexpr int n=250000, rounds=8;
  auto start=std::chrono::steady_clock::now();
  if(mode==0){
    for(int i=0;i<n;++i)d.push_back({uint64_t(i)%1000+1,uint64_t(i)});
    for(int i=0;i<n*rounds;++i){if((i&7)==0){d.pop_back();d.push_back({uint64_t(i)%1000+1,uint64_t(i)});}auto f=d.prod();hash+=f.a+f.b;}
  }else if(mode==1){
    for(int i=0;i<n/2;++i){d.push_front({uint64_t(i)%1000+1,uint64_t(i)});d.push_back({uint64_t(i)%1000+1,uint64_t(i)});}
    for(int i=0;i<n*rounds;++i){if((i&7)==0){d.pop_back();d.push_back({uint64_t(i)%1000+1,uint64_t(i)});}auto f=d.prod();hash+=f.a+f.b;}
  }else if(mode==2){
    for(int j=0;j<rounds;++j){for(int i=0;i<n;++i)d.push_back({uint64_t(i)%1000+1,uint64_t(i)});for(int i=0;i<n;++i){hash+=d.prod().b;d.pop_front();}}
  }else if(mode==3){
    for(int j=0;j<rounds;++j){for(int i=0;i<n;++i)d.push_front({uint64_t(i)%1000+1,uint64_t(i)});for(int i=0;i<n;++i){hash+=d.prod().b;d.pop_back();}}
  }else{
    for(int i=0;i<10000;++i)d.push_back({uint64_t(i)%1000+1,uint64_t(i)});
    for(int i=0;i<n*rounds;++i){auto r=rng();if(r%4==0)d.push_front({uint64_t(r)%1000+1,uint64_t(r)});else if(r%4==1)d.push_back({uint64_t(r)%1000+1,uint64_t(r)});else if(r%4==2)d.pop_front();else d.pop_back(); if((i&1)==0){auto f=d.prod();hash+=f.a+f.b;}}
  }
  auto end=std::chrono::steady_clock::now();
  std::cout<<std::chrono::duration<double,std::milli>(end-start).count()<<" "<<hash<<"\n";
}
