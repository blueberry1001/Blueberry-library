#!/usr/bin/env python3
"""Reproducible sparse-exp microbenchmarks; Linux/WSL, Python stdlib only."""
import argparse
import hashlib
import json
from pathlib import Path
import platform
import shutil
import statistics
import subprocess
import tempfile
import time


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--acl", type=Path, required=True, help="directory containing atcoder/")
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--compilers", nargs="+", default=["g++", "clang++"])
    parser.add_argument("--runs", type=int, default=3)
    parser.add_argument("--warmups", type=int, default=1)
    parser.add_argument("--n", type=int, default=500000)
    parser.add_argument("--timeout", type=float, default=120)
    parser.add_argument("--extra-flags", default="", help="additional compiler flags, shell splitting without execution")
    args = parser.parse_args()
    if args.runs < 1 or args.warmups < 0 or args.n <= 100:
        parser.error("need runs >= 1, warmups >= 0, n > 100")
    root = Path(__file__).resolve().parents[2]
    args.output.mkdir(parents=True, exist_ok=True)
    import shlex
    flags = ["-O2", "-DNDEBUG", "-std=gnu++20"] + shlex.split(args.extra_flags)
    tracked_sources = [Path(__file__).resolve(), Path(__file__).with_name("driver.cpp"),
                       root / "blueberry/math/sparse-exp.hpp", root / "blueberry/math/formal-power-series.hpp"]
    cpu = "unavailable"
    if Path("/proc/cpuinfo").exists():
        cpu = next((line.split(":", 1)[1].strip() for line in Path("/proc/cpuinfo").read_text().splitlines()
                    if line.startswith("model name")), cpu)
    meta = {"platform": platform.platform(), "python": platform.python_version(), "cpu": cpu,
            "flags": flags, "runs": args.runs, "warmups": args.warmups,
            "source_sha256": {str(p.relative_to(root)): digest(p) for p in tracked_sources},
            "git_head": subprocess.run(["git", "rev-parse", "HEAD"], cwd=root, text=True, capture_output=True).stdout.strip(),
            "acl": str(args.acl.resolve()), "compilers": {},
            "timing_scope": "in-memory text parse; complete API algorithm; to_chars format; external wall and peak RSS",
            "rss_method": "/usr/bin/time -f %M (KiB)"}
    acl_headers = [args.acl / "atcoder" / name for name in ["modint.hpp", "convolution.hpp", "internal_math.hpp"]]
    meta["acl_sha256"] = {str(p.relative_to(args.acl)): digest(p) for p in acl_headers if p.exists()}
    samples = []
    hashes = {}
    cases = [(args.n, k, distribution) for k in (1, 10, 100) for distribution in ("early", "late", "spread")]
    cases.append((500, 499, "early"))
    timer = Path("/usr/bin/time")
    if not timer.exists():
        raise SystemExit("/usr/bin/time is required for per-process peak RSS")
    raw_path = args.output / "samples.jsonl"
    # Do not silently overwrite old evidence.
    with raw_path.open("x", encoding="utf-8") as raw, tempfile.TemporaryDirectory(prefix="sparse-exp-") as tmp:
        tmp = Path(tmp)
        for compiler in args.compilers:
            executable = shutil.which(compiler)
            if not executable:
                meta["compilers"][compiler] = {"error": "not found"}
                continue
            version = subprocess.run([executable, "--version"], capture_output=True, text=True, check=True).stdout
            binary = tmp / (Path(compiler).name + "-bench")
            command = [executable, *flags, "-I", str(root), "-I", str(args.acl.resolve()),
                       str(Path(__file__).with_name("driver.cpp")), "-o", str(binary)]
            compiled = subprocess.run(command, capture_output=True, text=True)
            meta["compilers"][compiler] = {"version": version, "command": command,
                                           "compile_returncode": compiled.returncode, "compile_stderr": compiled.stderr}
            if compiled.returncode:
                continue
            meta["compilers"][compiler]["binary_sha256"] = digest(binary)
            for n, k, distribution in cases:
                case = f"n{n}-k{k}-{distribution}"
                # Round-robin methods within each run reduces order-dependent drift.
                for run in range(-args.warmups, args.runs):
                    for method in ("api", "table", "division", "batch"):
                        rss_path = tmp / "rss.txt"
                        command = [str(timer), "-f", "%M", "-o", str(rss_path), str(binary), method, str(n), str(k), distribution]
                        row = {"compiler": compiler, "case": case, "n": n, "k": k, "distribution": distribution,
                               "method": method, "run": run, "warmup": run < 0, "command": command}
                        start = time.perf_counter()
                        try:
                            proc = subprocess.run(command, capture_output=True, text=True, timeout=args.timeout)
                            row.update(wall_ms=(time.perf_counter() - start) * 1000, returncode=proc.returncode)
                            if proc.returncode:
                                row.update(error="execution failed", stderr=proc.stderr)
                            else:
                                row.update(json.loads(proc.stdout))
                                row["peak_rss_kib"] = int(rss_path.read_text().strip())
                                key = case
                                signature = row["input_hash"], row["output_hash"], row["output_bytes"]
                                if key in hashes and signature != hashes[key]:
                                    row["error"] = "input/output mismatch across methods, runs, or compilers"
                                hashes.setdefault(key, signature)
                        except (subprocess.TimeoutExpired, ValueError, OSError) as exc:
                            row["error"] = str(exc)
                        samples.append(row)
                        raw.write(json.dumps(row, sort_keys=True) + "\n")
                        raw.flush()
                print(f"{compiler}: {case}", flush=True)
    final_hashes = {str(p.relative_to(root)): digest(p) for p in tracked_sources}
    meta["sources_unchanged_during_run"] = final_hashes == meta["source_sha256"]
    if not meta["sources_unchanged_during_run"]:
        meta["source_sha256_after"] = final_hashes
    (args.output / "metadata.json").write_text(json.dumps(meta, indent=2) + "\n")
    groups = {}
    for row in samples:
        if not row["warmup"] and "error" not in row:
            groups.setdefault((row["compiler"], row["case"], row["method"]), []).append(row)
    summary = []
    for (compiler, case, method), rows in groups.items():
        entry = dict(compiler=compiler, case=case, method=method, samples=len(rows))
        for metric in ("parse_ms", "algorithm_ms", "format_ms", "normalize_ms", "inverse_ms", "recurrence_ms", "wall_ms", "peak_rss_kib"):
            values = [r[metric] for r in rows if metric in r]
            if values:
                entry[metric] = {"median": statistics.median(values), "min": min(values)}
        summary.append(entry)
    (args.output / "summary.json").write_text(json.dumps(summary, indent=2) + "\n")
    errors = [r for r in samples if "error" in r]
    if errors or not meta["sources_unchanged_during_run"] or any("error" in c or c.get("compile_returncode") != 0 for c in meta["compilers"].values()):
        raise SystemExit(f"benchmark incomplete: {len(errors)} failed/mismatched samples; see raw evidence")


if __name__ == "__main__":
    main()
