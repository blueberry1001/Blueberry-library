using E = vector<pair<int,int>>;
struct Result { uint64_t h = 1469598103934665603ULL, calls = 0; void operator()(const vector<int>& c) { ++calls; h = (h ^ c.size()) * 1099511628211ULL; for(int x:c) h = (h ^ uint64_t(x+1))*1099511628211ULL; } };
template<class Fn> auto collect(Fn f, int n, const E& edges) { vector<vector<int>> out; f(n,edges,[&](const vector<int>& c){out.push_back(c);}); return out; }
auto old_call = [](int n,const E& e,auto&& f) {base::enumerate_cliques(n,e,f);};
auto new_call = [](int n,const E& e,auto&& f) {candidate::enumerate_cliques(n,e,f);};
void check(int n,const E& e) {
 auto a=collect(old_call,n,e), b=collect(new_call,n,e); assert(a==b);
 vector<unsigned> got; for(auto c:b) { unsigned m=0; for(int x:c) m|=1u<<x; assert(__builtin_popcount(m)==int(c.size())); got.push_back(m); }
 sort(got.begin(),got.end()); assert(unique(got.begin(),got.end())==got.end());
 vector<unsigned> expected; for(unsigned m=1; m<(1u<<n); ++m) { bool ok=true; for(int i=0;i<n;++i) for(int j=i+1;j<n;++j) if((m>>i&1)&&(m>>j&1)&&find(e.begin(),e.end(),pair{i,j})==e.end()&&find(e.begin(),e.end(),pair{j,i})==e.end()) ok=false; if(ok)expected.push_back(m); }
 assert(got==expected);
}
void validate() {
 for(int n=0;n<=5;++n) { vector<pair<int,int>> all;for(int i=0;i<n;++i)for(int j=i+1;j<n;++j)all.emplace_back(i,j);for(unsigned mask=0;mask<(1u<<all.size());++mask){ E e;for(unsigned i=0;i<all.size();++i)if(mask>>i&1)e.push_back(all[i]);check(n,e); }}
 mt19937 rng(1729);for(int n=1;n<=10;++n)for(int t=0;t<100;++t){E e;for(int i=0;i<n;++i)for(int j=i+1;j<n;++j)if(rng()%5<2)e.emplace_back(i,j);shuffle(e.begin(),e.end(),rng);for(auto& [u,v]:e)if(rng()%2)swap(u,v);check(n,e);}
 E outer{{0,1},{0,2},{1,2}}, inner{{0,1}}; unsigned seen=0; candidate::enumerate_cliques(3,outer,[&](const vector<int>& c){ assert(!c.empty()); vector<vector<int>> saved; candidate::enumerate_cliques(2,inner,[&](const vector<int>& x){saved.push_back(x);});assert(saved.size()==3);++seen;});assert(seen==7);
 bool caught=false; try { candidate::enumerate_cliques(3,outer,[&](const vector<int>&){throw runtime_error("expected");}); } catch(const runtime_error&) { caught=true; } assert(caught);
 cout<<"ok\n";
}

 if(argc==1){validate();return 0;}

 if(shape==0){n=17;for(int i=0;i<n;++i)for(int j=i+1;j<n;++j)e.emplace_back(i,j);}
 if(shape==1){n=110;for(int i=0;i<n;++i)for(int j=i+1;j<n;++j)if(rng()%4==0)e.emplace_back(i,j);}
