import hashlib
import json
import os
from pathlib import Path
import subprocess

root = Path(__file__).resolve().parents[4]
work = Path(__file__).resolve().parent
vc = Path(r'C:\Program Files (x86)\Microsoft Visual Studio\2022\BuildTools\VC\Tools\MSVC\14.43.34808')
sdk = Path(r'C:\Program Files (x86)\Windows Kits\10')
version = '10.0.22621.0'
compiler = Path(r'C:\Program Files\Unity\Hub\Editor\6000.5.2f1\Editor\Data\PlaybackEngines\AndroidPlayer\NDK\toolchains\llvm\prebuilt\windows-x86_64\bin\clang++.exe')
env = os.environ.copy()
env['PATH'] = str(vc / 'bin/Hostx64/x64') + os.pathsep + env['PATH']
env['INCLUDE'] = os.pathsep.join(map(str, [vc / 'include', *(sdk / 'Include' / version / p for p in ('ucrt', 'shared', 'um'))]))
env['LIB'] = os.pathsep.join(map(str, [vc / 'lib/x64', *(sdk / 'Lib' / version / p / 'x64' for p in ('ucrt', 'um'))]))
records = []
for std in ('gnu++20', 'gnu++23'):
    exe = work / ('test-' + std + '.exe')
    command = [str(compiler), '--target=x86_64-pc-windows-msvc', '-fuse-ld=link', '-std='+std, '-O2', '-Wall', '-Wextra', '-Wshadow', '-Werror', '-iquote', str(root), '-isystem', str(root / '.deps/ac-library'), str(root / 'tests/random/algebra-expansion.cpp'), '-o', str(exe)]
    build = subprocess.run(command, env=env, capture_output=True, text=True)
    sources = [root / 'blueberry/math/subset-convolution.hpp', root / 'tests/random/algebra-expansion.cpp', root / '.deps/ac-library/atcoder/modint.hpp']
    entry = {'command':command, 'source_sha256':{str(p.relative_to(root)):hashlib.sha256(p.read_bytes()).hexdigest() for p in sources}, 'returncode':build.returncode, 'stdout':build.stdout, 'stderr':build.stderr, 'runs':[]}
    records.append(entry)
    if build.returncode == 0:
        for seed in (1,17,731):
            test = subprocess.run([str(exe), str(seed)], capture_output=True, text=True)
            entry['runs'].append({'seed':seed, 'returncode':test.returncode, 'stdout':test.stdout, 'stderr':test.stderr})
            if test.returncode: break
    if build.returncode or any(x['returncode'] for x in entry['runs']): break
(work / 'test.json').write_text(json.dumps(records, indent=2)+'\n', encoding='utf-8')
if len(records) != 2 or any(x['returncode'] or len(x['runs']) != 3 or any(y['returncode'] for y in x['runs']) for x in records): raise SystemExit(1)
print('PASS: C++20/23, seeds 1/17/731')
