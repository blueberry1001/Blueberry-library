import hashlib
import json
import os
import ctypes
from pathlib import Path
import platform
import statistics
import subprocess

here = Path(__file__).resolve().parent
vc = Path(r'C:\Program Files (x86)\Microsoft Visual Studio\2022\BuildTools\VC\Tools\MSVC\14.43.34808')
sdk = Path(r'C:\Program Files (x86)\Windows Kits\10')
version = '10.0.22621.0'
msvc = vc / 'bin/Hostx64/x64/cl.exe'
clang = Path(r'C:\Program Files\Unity\Hub\Editor\6000.5.2f1\Editor\Data\PlaybackEngines\AndroidPlayer\NDK\toolchains\llvm\prebuilt\windows-x86_64\bin\clang++.exe')
env = os.environ.copy()
env['PATH'] = str(msvc.parent) + os.pathsep + env.get('PATH', '')
env['INCLUDE'] = os.pathsep.join(map(str, [vc/'include', *(sdk/'Include'/version/p for p in ('ucrt', 'shared', 'um'))]))
env['LIB'] = os.pathsep.join(map(str, [vc/'lib/x64', *(sdk/'Lib'/version/p/'x64' for p in ('ucrt', 'um'))]))
def cpu_times():
  idle=ctypes.c_ulonglong();kernel=ctypes.c_ulonglong();user=ctypes.c_ulonglong()
  if not ctypes.windll.kernel32.GetSystemTimes(ctypes.byref(idle),ctypes.byref(kernel),ctypes.byref(user)): return None
  return idle.value, kernel.value, user.value
out = {'status':'running', 'platform':platform.platform(), 'source_sha256':hashlib.sha256((here/'benchmark.cpp').read_bytes()).hexdigest(), 'variants':['baseline','hoist_div','mul_loop','mul_sqrt_newton','mul_power_newton','mul_power_sqrt_binary','fixed_bits','nyaan','bounded_bits_mul','bounded_bits_power','newton_small_binary','newton_small_bits'], 'note':'Nyaan excluded on MSVC without __int128; other sessions may be active; CPU not pinned.', 'rows':[]}
def save(): (here/'results.json').write_text(json.dumps(out, indent=2)+'\n')
for compiler in ['msvc','clang']:
  exe = here/(compiler+'.exe')
  if compiler == 'msvc': cmd = [str(msvc),'/nologo','/std:c++20','/EHsc','/O2','/DNDEBUG',str(here/'benchmark.cpp'),'/Fo'+str(here/'msvc.obj'),'/Fe'+str(exe)]
  else: cmd = [str(clang),'--target=x86_64-pc-windows-msvc','-fuse-ld=link','-std=gnu++20','-O2','-DNDEBUG',str(here/'benchmark.cpp'),'-o',str(exe)]
  p = subprocess.run(cmd,env=env,capture_output=True,text=True,errors='replace')
  out['rows'].append({'compiler':compiler,'compile':cmd,'compile_exit':p.returncode,'compile_output':p.stdout+p.stderr,'samples':[]})
  row=out['rows'][-1]; save()
  if p.returncode: out['status']='failed';save();raise SystemExit(p.returncode)
  for shape in range(8):
    checksums = {}
    variants=list(range(7))+[8,9,10,11] if compiler=='msvc' else list(range(12))
    for rep in range(-1,5):
      for i in range(len(variants)):
        variant=variants[(i+rep+len(variants))%len(variants)]
        before=cpu_times()
        p=subprocess.run([str(exe),str(variant),str(shape),'50000'],env=env,capture_output=True,text=True,errors='replace',timeout=120)
        after=cpu_times()
        if p.returncode: out['status']='failed';row['error']=p.stdout+p.stderr;save();raise SystemExit(p.returncode)
        v,s,count,ms,checksum=p.stdout.strip().split()
        total=after[1]+after[2]-before[1]-before[2]
        load=None if not total else 100*(1-(after[0]-before[0])/total)
        row['samples'].append({'shape':shape,'rep':rep,'variant':variant,'ms':float(ms),'checksum':checksum,'system_cpu_pct':load})
        checksums.setdefault(checksum,[]).append(variant)
      if len(checksums)!=1: out['status']='mismatch';row['checksums']=checksums;save();raise SystemExit(1)
      save()
    print(compiler,'shape',shape,'done',flush=True)
out['status']='complete';save()
for row in out['rows']:
  print(row['compiler'])
  for shape in range(8):
    variants=list(range(7))+[8,9,10,11] if row['compiler']=='msvc' else list(range(12))
