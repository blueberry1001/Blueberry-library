using U = std::uint64_t;
using Clock = std::chrono::steady_clock;

U baseline(U n, int k) {
  if (k == 1 || n <= 1) return n;
  if (k >= 64) return 1;
  auto fits = [n, k](U x) {
    U v = 1;
    for (int i = 0; i < k; ++i) { if (v > n / x) return false; v *= x; }
    return true;
  };
  U lo = 1, hi = U{1} << ((std::bit_width(n) + k - 1) / k);
  while (hi - lo > 1) { U mid = lo + (hi - lo) / 2; if (fits(mid)) lo = mid; else hi = mid; }
  return lo;
}

template <int mode>
U candidate(U n, int k) {
  if (k == 1 || n <= 1) return n;
  const int width = int(std::bit_width(n));
  if (k >= width) return 1;
  if constexpr (mode >= 2) {
    if (k == 2) {
      U x = U{1} << ((width + 1) / 2);
      while (true) { U y = (x + n / x) / 2; if (y >= x) return x; x = y; }
    }
  }
  U lo = 1, hi = U{1} << ((width + k - 1) / k);
  while (hi - lo > 1) {
    const U x = lo + (hi - lo) / 2;
    U v = 1;
    bool good = true;
    if constexpr (mode == 0) {
      const U bound = n / x;
      for (int i = 0; i < k; ++i) { if (v > bound) { good = false; break; } v *= x; }
    } else if constexpr (mode == 1 || mode == 2) {
      for (int i = 0; i < k; ++i) {
#if defined(__GNUC__) || defined(__clang__)
        if (__builtin_mul_overflow(v, x, &v) || v > n) { good = false; break; }
#else
        if (v > n / x) { good = false; break; } v *= x;
#endif
      }
    } else if constexpr (mode == 3 || mode == 4) {
      U y = x;
      int e = k;
      while (e) {
        if (e & 1) {
#if defined(__GNUC__) || defined(__clang__)
          if (__builtin_mul_overflow(v, y, &v) || v > n) { good = false; break; }
#else
          if (v > n / y) { good = false; break; } v *= y;
#endif
        }
        e >>= 1;
        if (!e) break;
#if defined(__GNUC__) || defined(__clang__)
        if (__builtin_mul_overflow(y, y, &y) || y > n) { good = false; break; }
#else
        if (y > n / y) { good = false; break; } y *= y;
#endif
      }
    }
    if (good) lo = x; else hi = x;
  }
  return lo;
}

U integer_sqrt_binary(U n, int k) {
  if (k != 2 || n <= 1) return candidate<3>(n, k);
  U lo = 1, hi = U{1} << ((std::bit_width(n) + 1) / 2);
  while (hi - lo > 1) { U m = lo + (hi - lo) / 2; if (m * m <= n) lo = m; else hi = m; }
  return lo;
}

U fixed_bits(U n, int k) {
  if (k == 1 || n <= 1) return n;
  if (k >= 64) return 1;
  U ret = 0;
  for (int bit = 31; bit >= 0; --bit) {
    const U x = ret | (U{1} << bit);
    U v = 1;
    bool good = true;
    for (int i = 0; i < k; ++i) {
#if defined(__GNUC__) || defined(__clang__)
      if (__builtin_mul_overflow(v, x, &v)) { good = false; break; }
#else
      if (v > std::numeric_limits<U>::max() / x) { good = false; break; } v *= x;
#endif
    }
    if (good && v <= n) ret = x;
  }
  return ret;
}

U nyaan(U n, int k) {
#if defined(__SIZEOF_INT128__)
  return kth_root_integral(n, static_cast<U>(k));
#else
  return baseline(n, k);
#endif
}

template <bool power>
U bounded_bits(U n, int k) {
  if (k == 1 || n <= 1) return n;
  const int width = int(std::bit_width(n));
  if (k >= width) return 1;
  if (k == 2) {
    U x = U{1} << ((width + 1) / 2);
    while (true) { U y = (x + n / x) / 2; if (y >= x) return x; x = y; }
  }
  U root = 0;
  for (int bit = (width - 1) / k; bit >= 0; --bit) {
    U x = root | (U{1} << bit), v = 1;
    bool good = true;
    if constexpr (!power) {
      for (int i = 0; i < k; ++i) {
#if defined(__GNUC__) || defined(__clang__)
        if (__builtin_mul_overflow(v, x, &v) || v > n) { good = false; break; }
#else
        if (v > n / x) { good = false; break; } v *= x;
#endif
      }
    } else {
      int e = k;
      while (e) {
        if (e & 1) {
#if defined(__GNUC__) || defined(__clang__)
          if (__builtin_mul_overflow(v, x, &v) || v > n) { good = false; break; }
#else
          if (v > n / x) { good = false; break; } v *= x;
#endif
        }
        e >>= 1;
        if (!e) break;
#if defined(__GNUC__) || defined(__clang__)
        if (__builtin_mul_overflow(x, x, &x) || x > n) { good = false; break; }
#else
        if (x > n / x) { good = false; break; } x *= x;
#endif
      }
    }
    if (good) root |= U{1} << bit;
  }
  return root;
}

int main(int argc, char** argv) {
  if (argc != 4) return 2;
  const int which = std::atoi(argv[1]), shape = std::atoi(argv[2]), count = std::atoi(argv[3]);
  U (*fn[])(U, int) = {baseline, candidate<0>, candidate<1>, candidate<2>, candidate<3>, integer_sqrt_binary, fixed_bits, nyaan, bounded_bits<false>, bounded_bits<true>};
  std::mt19937_64 rng(20261118 + shape);
  std::vector<std::pair<U,int>> input(count);
  for (int i = 0; i < count; ++i) {
    U n = rng();
    int k = shape == 0 ? 2 : shape == 1 ? 3 : shape == 2 ? 4 : shape == 3 ? 7 : shape == 4 ? 16 : shape == 5 ? 63 : shape == 6 ? 1 + int(rng() % 65) : 2 + int(rng() % 15);
    if (shape == 7) n = rng() & ((U{1} << 32) - 1);
    input[i] = {n, k};
  }
  U sum = 0;
  auto t = Clock::now();
  for (const auto& [n,k] : input) sum += fn[which](n,k);
  auto e = Clock::now();
  std::printf("%d %d %d %.6f %llu\n", which, shape, count, std::chrono::duration<double,std::milli>(e-t).count(), static_cast<unsigned long long>(sum));
}
