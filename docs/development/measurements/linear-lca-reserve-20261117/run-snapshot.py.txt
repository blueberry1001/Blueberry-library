import argparse
import hashlib
import json
import os
from pathlib import Path
import platform
import shutil
import statistics
import subprocess
import time

root = Path(__file__).resolve().parents[1]
parser = argparse.ArgumentParser()
parser.add_argument('--prepare', action='store_true')
parser.add_argument('--output', type=Path, required=True)
args = parser.parse_args()
build = root / '.build/linear-lca-reserve'
vc = Path(r'C:\Program Files (x86)\Microsoft Visual Studio\2022\BuildTools\VC\Tools\MSVC\14.43.34808')
sdk = Path(r'C:\Program Files (x86)\Windows Kits\10')
ver = '10.0.22621.0'
compiler = vc / 'bin/Hostx64/x64/cl.exe'
env = os.environ.copy()
env['PATH'] = str(compiler.parent) + os.pathsep + env.get('PATH', '')
env['INCLUDE'] = os.pathsep.join(map(str, [vc / 'include', *(sdk / 'Include' / ver / p for p in ('ucrt', 'shared', 'um'))]))
env['LIB'] = os.pathsep.join(map(str, [vc / 'lib/x64', *(sdk / 'Lib' / ver / p / 'x64' for p in ('ucrt', 'um'))]))
source = root / 'blueberry/graph/linear-lca.hpp'
rmq = root / 'blueberry/data-structure/linear-rmq.hpp'
driver = root / 'benchmark/linear-lca-reserve.cpp'
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
result = {'timestamp_utc': time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime()), 'platform': platform.platform(), 'processor': platform.processor(), 'compiler': str(compiler), 'flags': ['/std:c++20', '/EHsc', '/O2', '/DNDEBUG'], 'warmups': 1, 'runs': 7, 'seed': 1729, 'source_sha256': sha(source), 'rmq_sha256': sha(rmq), 'driver_sha256': sha(driver), 'runner_sha256': sha(Path(__file__)), 'rss': 'not measured', 'samples': [], 'commands': []}
args.output.parent.mkdir(parents=True, exist_ok=True)
def save():
    args.output.write_text(json.dumps(result, indent=2) + '\n', encoding='utf-8')
try:
    for name in ('baseline', 'reserve'):
        target = build / name
        graph = target / 'blueberry/graph'
        ds = target / 'blueberry/data-structure'
        graph.mkdir(parents=True, exist_ok=True)
        ds.mkdir(parents=True, exist_ok=True)
        header = source.read_text(encoding='utf-8')
        needle = '    depth_[root] = 0;'
        if name == 'reserve':
            assert header.count(needle) == 1
            header = header.replace(needle, '    tour_.reserve(2 * tree.size() - 1); depths.reserve(2 * tree.size() - 1);\n' + needle)
        (graph / 'linear-lca.hpp').write_text(header, encoding='utf-8')
        shutil.copyfile(rmq, ds / 'linear-rmq.hpp')
        cmd = [str(compiler), '/nologo', *result['flags'], '/I' + str(target), str(driver), '/Fo' + str(target / 'driver.obj'), '/Fe' + str(target / 'driver.exe')]
        p = subprocess.run(cmd, cwd=root, env=env, capture_output=True, text=True, errors='replace')
        result['commands'].append({'command': cmd, 'exit': p.returncode, 'stdout': p.stdout, 'stderr': p.stderr})
        if p.returncode: raise RuntimeError('compile failed')
    if args.prepare:
        result['status'] = 'prepared'
    else:
        for shape in ('random', 'path', 'star'):
            checksum = None
            for run in range(-1, 7):
                variants = ('baseline', 'reserve') if run % 2 else ('reserve', 'baseline')
                for name in variants:
                    cmd = [str(build / name / 'driver.exe'), '500000', shape]
                    p = subprocess.run(cmd, cwd=root, capture_output=True, text=True, errors='replace')
                    if p.returncode: raise RuntimeError(f'{cmd}: {p.returncode}: {p.stderr}')
                    fields = p.stdout.split()
                    if len(fields) != 3 or (checksum is not None and fields[0] != checksum): raise RuntimeError('checksum mismatch: ' + p.stdout)
                    checksum = fields[0]
                    if run >= 0:
                        result['samples'].append({'shape': shape, 'run': run + 1, 'variant': name, 'checksum': checksum, 'build_ms': float(fields[1]), 'query_ms': float(fields[2])})
                        save()
            for name in ('baseline', 'reserve'):
                rows = [x for x in result['samples'] if x['shape'] == shape and x['variant'] == name]
                print(shape, name, 'build', statistics.median(x['build_ms'] for x in rows), 'query', statistics.median(x['query_ms'] for x in rows), flush=True)
        result['status'] = 'complete'
    save()
except BaseException as exc:
    result['status'] = 'failed'
    result['error'] = repr(exc)
    save()
    raise
