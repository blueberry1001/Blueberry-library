template<class S> Mint oracle(const S& s){std::map<int,Mint> prev;Mint ans=1;for(auto v:s){Mint next=ans+ans-prev[static_cast<int>(v)];prev[static_cast<int>(v)]=ans;ans=next;}return ans-1;}

struct Symbol {int n; bool operator<(const Symbol& b)const{return n/3<b.n/3;}};
