#pragma once

#include <algorithm>
#include <cstddef>
#include <cstdint>
#include <limits>
#include <numeric>
#include <type_traits>
#include <vector>

namespace blueberry {

template <class Count, class Sequence>
Count count_subsequences(const Sequence& sequence) {
  const std::size_t n = sequence.size();
  using Symbol = std::remove_cvref_t<decltype(sequence[std::size_t{}])>;
  if constexpr (std::is_integral_v<Symbol> && sizeof(Symbol) == 1 && sizeof(Count) <= 1024 && std::numeric_limits<unsigned char>::digits == 8) {
    if (n >= 64) {
      std::vector<Count> previous(256, Count(0));
      Count total(1);
      for (std::size_t i = 0; i < n; ++i) {
        auto symbol = static_cast<unsigned char>(sequence[i]);
        const Count next = total + total - previous[symbol];
        previous[symbol] = total; total = next;
      }
      return total - Count(1);
    }
  }
  if constexpr (std::is_integral_v<Symbol> && !std::is_same_v<Symbol, bool> && sizeof(Symbol) > 1 && sizeof(Symbol) <= sizeof(std::uintmax_t)) {
    if (n >= 256) {
      using U = std::make_unsigned_t<Symbol>;
      Symbol low = sequence[0], high = low;
      const std::uintmax_t limit = std::min({std::size_t(65535), n / 2, std::size_t(262144) / sizeof(Count)});
      bool narrow = true;
      for (std::size_t k = 1; k <= 8; ++k) {
        Symbol x = sequence[k == 8 ? n - 1 : n / 8 * k];
        if (x < low) low = x;
        if (x > high) high = x;
        if (static_cast<std::uintmax_t>(static_cast<U>(U(high) - U(low))) > limit) { narrow = false; break; }
      }
      for (std::size_t i = 1; narrow && i < n; ++i) {
        Symbol x = sequence[i];
        if (x < low) low = x;
        if (x > high) high = x;
        if (static_cast<std::uintmax_t>(static_cast<U>(U(high) - U(low))) > limit) narrow = false;
      }
      if (narrow) {
        const auto width = static_cast<std::size_t>(static_cast<U>(U(high) - U(low))) + 1;
        std::vector<Count> previous(width, Count(0));
        Count total(1);
        for (std::size_t j = 0; j < n; ++j) {
          const auto rank = static_cast<std::size_t>(static_cast<U>(U(sequence[j]) - U(low)));
          const Count next = total + total - previous[rank];
          previous[rank] = total; total = next;
        }
        return total - Count(1);
      }
    }
  }
  std::vector<std::size_t> order(n), rank(n);
  std::iota(order.begin(), order.end(), std::size_t{0});
  std::sort(order.begin(), order.end(), [&](auto i, auto j) {
    return sequence[i] < sequence[j];
  });
  std::size_t kinds = 0;
  for (std::size_t i = 0; i < n; ++i) {
    if (i == 0 || sequence[order[i - 1]] < sequence[order[i]]) ++kinds;
    rank[order[i]] = kinds - 1;
  }
  std::vector<Count> previous(kinds, Count(0));
  Count total(1);
  for (auto symbol : rank) {
    const Count next = total + total - previous[symbol];
    previous[symbol] = total;
    total = next;
  }
  return total - Count(1);
}

}
