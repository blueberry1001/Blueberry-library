using namespace std;
using Edge = pair<int,int>;
using Tri = array<int,3>;
void check(int n, const vector<Edge>& edges) {
  vector<Tri> old, now, expected;
  blueberry::enumerate_triangles(n, edges, [&](int a,int b,int c){old.push_back({a,b,c});});
  candidate::enumerate_triangles(n, edges, [&](int a,int b,int c){now.push_back({a,b,c});});
  assert(old == now);
  vector<vector<bool>> adj(n, vector<bool>(n));
  for (auto [u,v] : edges) adj[u][v] = adj[v][u] = true;
  for (int a=0;a<n;++a) for (int b=a+1;b<n;++b) for (int c=b+1;c<n;++c)
    if (adj[a][b] && adj[b][c] && adj[a][c]) expected.push_back({a,b,c});
  for (auto& t : now) sort(t.begin(),t.end());
  sort(now.begin(),now.end()); assert(now==expected);
}
int main(int argc,char** argv) {
  if (argc==1) {
    mt19937 gen(1729);
    for (int n=0;n<=5;++n) {
      vector<Edge> all;
      for(int u=0;u<n;++u) for(int v=u+1;v<n;++v) all.emplace_back(u,v);
      for(unsigned mask=0;mask<(1u<<all.size());++mask) {
        vector<Edge> e; for(unsigned i=0;i<all.size();++i) if(mask>>i&1) e.push_back(all[i]);
        check(n,e);
      }
    }
    for (int n=0;n<=15;++n) for(int t=0;t<100;++t) {
      vector<Edge> e;
      for(int u=0;u<n;++u) for(int v=u+1;v<n;++v) if(gen()%5<2) e.emplace_back(u,v);
      shuffle(e.begin(),e.end(),gen); for(auto& [u,v]:e) if(gen()&1) swap(u,v); check(n,e);
    }
    cout<<"ok\n"; return 0;
  }
  int which=atoi(argv[1]),shape=atoi(argv[2]),reps=atoi(argv[3]),n=0;
  vector<Edge> edges;
  mt19937 gen(1949);
  if(shape==0) { n=100000; }
  if(shape==1) { n=100000; for(int i=1;i<n;++i) edges.emplace_back(0,i); }
  if(shape==2) { n=100000; for(int i=1;i<n;++i) edges.emplace_back(i-1,i); }
  if(shape==3) { n=1800; for(int u=0;u<n;++u) for(int v=u+1;v<n;++v) if(gen()%25==0) edges.emplace_back(u,v); }
  if(shape==4) { n=350; for(int u=0;u<n;++u) for(int v=u+1;v<n;++v) edges.emplace_back(u,v); }

  shuffle(edges.begin(),edges.end(),gen);
  uint64_t hash=0,count=0;
  auto emit=[&](int a,int b,int c) { hash=hash*0x9e3779b185ebca87ULL + uint64_t(a+1)*101 + uint64_t(b+1)*17 + uint64_t(c+1); ++count; };
  auto begin=chrono::steady_clock::now();

  auto ms=chrono::duration<double,milli>(chrono::steady_clock::now()-begin).count();

  cout<<ms<<' '<<hash<<' '<<count<<' '<<n<<' '<<edges.size()<<' '<<counters.PeakWorkingSetSize<<'\n';
