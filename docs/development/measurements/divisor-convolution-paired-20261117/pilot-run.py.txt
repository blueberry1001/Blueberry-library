ctypes.windll.kernel32.GetCurrentProcess.restype = ctypes.c_void_p
ctypes.windll.kernel32.GetProcessAffinityMask.argtypes = [ctypes.c_void_p, ctypes.POINTER(ctypes.c_size_t), ctypes.POINTER(ctypes.c_size_t)]
ctypes.windll.kernel32.SetProcessAffinityMask.argtypes = [ctypes.c_void_p, ctypes.c_size_t]
proc=ctypes.windll.kernel32.GetCurrentProcess(); mask=ctypes.c_size_t(); system=ctypes.c_size_t()
if ctypes.windll.kernel32.GetProcessAffinityMask(proc,ctypes.byref(mask),ctypes.byref(system)):
    print('affinity',hex(mask.value & -mask.value),'set',ctypes.windll.kernel32.SetProcessAffinityMask(proc,mask.value & -mask.value),flush=True)
else: print('affinity unavailable',flush=True)
for n,repeat in [(20,20000),(1000,1000),(100000,15),(1000000,2)]:
    for modes in ((0,1),(3,4)):
        values={m:[] for m in modes}; hashes={}
        for sample in range(6):
            for m in (modes if sample%2==0 else modes[::-1]):
                out=subprocess.check_output([str(exe),str(m),str(n),str(repeat)],text=True).split()
                hashes.setdefault(m,set()).add(out[1]); val=float(out[0]); print('sample',n,repeat,m,sample,val,out[1],flush=True)
                if sample: values[m].append(val)
        if len(set.union(*hashes.values()))!=1: raise RuntimeError('checksum mismatch')
        print('summary',n,repeat,modes,'median',*[statistics.median(values[m]) for m in modes],'range',*[str((min(values[m]),max(values[m]))) for m in modes],flush=True)
