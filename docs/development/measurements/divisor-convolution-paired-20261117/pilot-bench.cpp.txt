struct Mint {
  uint32_t x = 0;
  static constexpr uint32_t mod = 998244353;
  Mint() = default;
  explicit Mint(uint32_t a) : x(a) {}
  Mint& operator+=(Mint a) { x += a.x; if (x >= mod) x -= mod; return *this; }
  Mint& operator-=(Mint a) { x += mod - a.x; if (x >= mod) x -= mod; return *this; }
  Mint& operator*=(Mint a) { x = uint64_t(x) * a.x % mod; return *this; }
  bool operator==(const Mint&) const = default;
};
template<bool Divisor, bool Inverse, bool Paired, class T>
void transform(vector<T>& a, vector<T>* b = nullptr) {
  size_t n = a.size(); vector<bool> composite(n);
  for (size_t p = 2; p < n; ++p) {
    if (composite[p]) continue;
    size_t count = (n - 1) / p;
    for (size_t j = 1; j <= count; ++j) {
      composite[j * p] = true;
      size_t i = (Divisor != Inverse) ? j : count + 1 - j;
      if constexpr (Divisor) {
        if constexpr (Inverse) { a[i*p] -= a[i]; if constexpr(Paired) (*b)[i*p] -= (*b)[i]; }
        else { a[i*p] += a[i]; if constexpr(Paired) (*b)[i*p] += (*b)[i]; }
      } else {
        if constexpr (Inverse) { a[i] -= a[i*p]; if constexpr(Paired) (*b)[i] -= (*b)[i*p]; }
        else { a[i] += a[i*p]; if constexpr(Paired) (*b)[i] += (*b)[i*p]; }
      }
    }
  }
}
template<bool Divisor, class T> vector<T> conv(vector<T> a, vector<T> b) {
  transform<Divisor,false,true>(a,&b);
  for (size_t i=1;i<a.size();++i) a[i] *= b[i];
  transform<Divisor,true,false>(a);
  if (!a.empty()) a[0] = T{};
  return a;
}
