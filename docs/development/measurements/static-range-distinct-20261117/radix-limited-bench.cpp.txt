#include <algorithm>
#include <chrono>
#include <cstdint>
#include <cstdlib>
#include <iostream>
#include <limits>
#include <random>
#include <string>
#include <utility>
#include <vector>
#define NOMINMAX
#include <windows.h>
#include <psapi.h>
#pragma comment(lib, "psapi.lib")
#include "blueberry/data-structure/static-range-distinct.hpp"

#ifdef BENCH64
using Value = std::int64_t;
#else
using Value = int;
#endif

int main(int argc, char** argv) {
  const int n = std::atoi(argv[1]);
  const std::string mode = argv[2];
  std::mt19937_64 rng(20261117);
  std::vector<Value> a(n);
  for (int i = 0; i < n; ++i) {
    if (mode == "equal") a[i] = 0;
    else if (mode == "eight") a[i] = rng() % 8;
    else if (mode == "256") a[i] = rng() % 256;
    else if (mode == "513") a[i] = rng() % 513;
    else if (mode == "2048") a[i] = rng() % 2048;
    else if (mode == "2049") a[i] = rng() % 2049;
    else if (mode == "4096") a[i] = rng() % 4096;
    else if (mode == "late2049") a[i] = i + 1 == n ? 2048 : i < 2048 ? i : rng() % 2048;
    else if (mode == "half2049") a[i] = i == n / 2 ? 2048 : i < 2048 ? i : rng() % 2048;
    else if (mode == "early2049") a[i] = i == 2048 ? 2048 : i < 2048 ? i : rng() % 2048;
    else if (mode == "window2049") a[i] = i == n / 8 - 1 ? 2048 : i < 2048 ? i : rng() % 2048;
    else if (mode == "after2049") a[i] = i == n / 8 ? 2048 : i < 2048 ? i : rng() % 2048;
    else if (mode == "boundary") {
      Value values[] = {std::numeric_limits<Value>::min(), std::numeric_limits<Value>::max(), 0, -1, 1};
      a[i] = values[rng() % 5];
    }
    else if (mode == "ascending") a[i] = i;
    else if (mode == "reverse") a[i] = n - i;
    else a[i] = static_cast<Value>(rng());
  }
  std::vector<std::pair<int,int>> queries(20000);
  for (auto& [l,r] : queries) { l = rng() % (n + 1); r = rng() % (n + 1); if (l > r) std::swap(l,r); }
  auto before = std::chrono::steady_clock::now();
  blueberry::StaticRangeDistinct<Value> d(a);
  auto built = std::chrono::steady_clock::now();
  std::uint64_t checksum = 0;
  for (auto [l,r] : queries) checksum = checksum * 0x9e3779b97f4a7c15ULL + d.count(l,r);
  auto done = std::chrono::steady_clock::now();
  PROCESS_MEMORY_COUNTERS mem{};
  auto ok = GetProcessMemoryInfo(GetCurrentProcess(), &mem, sizeof(mem));
  std::cout << std::chrono::duration<double, std::milli>(built-before).count() << ' '
            << std::chrono::duration<double, std::milli>(done-built).count() << ' '
            << checksum << ' ' << (ok ? mem.PeakWorkingSetSize : 0) << '\n';
}
