import hashlib, json, os, subprocess
from pathlib import Path

root = Path.cwd()
out = root / '.build/trie-erase-study'
source = (root / 'blueberry/data-structure/binary-trie.hpp').read_text(encoding='utf-8')
old = '''    if (!count(value)) return false;
    std::array<int, Bits + 1> path{}; int v = root_; path[0] = v;
    for (int bit = Bits - 1; bit >= 0; --bit) { v = nodes_[v].child[(value >> bit) & UInt{1}]; path[Bits - bit] = v; }
'''
new = '''    if (root_ == -1) return false;
    std::array<int, Bits + 1> path; int v = root_; path[0] = v;
    for (int bit = Bits - 1; bit >= 0; --bit) {
      v = nodes_[v].child[(value >> bit) & UInt{1}];
      if (v == -1) return false;
      path[Bits - bit] = v;
    }
'''
assert source.count(old) == 1
candidate = source.replace(old, new)
vc = Path(r'C:\Program Files (x86)\Microsoft Visual Studio\2022\BuildTools\VC\Tools\MSVC\14.43.34808')
sdk = Path(r'C:\Program Files (x86)\Windows Kits\10'); ver = '10.0.22621.0'
compiler = Path(r'C:\Program Files\Unity\Hub\Editor\6000.5.2f1\Editor\Data\PlaybackEngines\AndroidPlayer\NDK\toolchains\llvm\prebuilt\windows-x86_64\bin\clang++.exe')
env = os.environ.copy(); env['PATH'] = str(vc/'bin/Hostx64/x64') + os.pathsep + env['PATH']
env['INCLUDE'] = os.pathsep.join(map(str, [vc/'include', *(sdk/'Include'/ver/p for p in ('ucrt','shared','um'))]))
env['LIB'] = os.pathsep.join(map(str, [vc/'lib/x64', *(sdk/'Lib'/ver/p/'x64' for p in ('ucrt','um'))]))
records=[]
for name, content in [('baseline',source),('candidate',candidate)]:
    dest=out/name; hdr=dest/'blueberry/data-structure/binary-trie.hpp'; hdr.parent.mkdir(parents=True,exist_ok=True)
    hdr.write_text(content,encoding='utf8',newline='\n')
    for target, src, standard in [('bench',out/'bench.cpp','gnu++20'),('test20',root/'tests/random/operation-structures.cpp','gnu++20'),('test23',root/'tests/random/operation-structures.cpp','gnu++23')]:
        cmd=[str(compiler),'--target=x86_64-pc-windows-msvc','-fuse-ld=link','-std='+standard,'-O2', '-DNDEBUG' if target=='bench' else '-UNDEBUG', '-I'+str(dest),'-I'+str(root),str(src),'-o',str(dest/(target+'.exe'))]
        p=subprocess.run(cmd,env=env,capture_output=True,text=True,errors='replace')
        records.append({'variant':name,'target':target,'command':cmd,'returncode':p.returncode,'stderr':p.stderr,'source_sha256':hashlib.sha256(content.encode()).hexdigest()})
        (out/'compile.json').write_text(json.dumps(records,indent=2)+'\n',encoding='utf8')
        if p.returncode: raise SystemExit('compile failed')
        if target.startswith('test'):
            for seed in (1,17,731):
                p=subprocess.run([str(dest/(target+'.exe')),str(seed)],capture_output=True,text=True,errors='replace',timeout=120)
                records.append({'variant':name,'target':target,'seed':seed,'returncode':p.returncode,'stderr':p.stderr})
                (out/'compile.json').write_text(json.dumps(records,indent=2)+'\n',encoding='utf8')
                if p.returncode: raise SystemExit('test failed')
print('prepared')
