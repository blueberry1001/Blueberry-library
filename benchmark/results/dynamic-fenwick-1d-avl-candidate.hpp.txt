#pragma once

#include <algorithm>
#include <cassert>
#include <cstddef>
#include <limits>
#include <type_traits>
#include <utility>
#include <vector>

namespace blueberry {

// Experimental benchmark candidate: augmented AVL over updated coordinates.
// This prototype deliberately supports long long only; it is not a public API.
template <class T, class Coord = long long>
class DynamicFenwickTree {
  static_assert(std::is_same_v<T, long long> && std::is_same_v<Coord, long long>);

  struct Node {
    Coord key;
    T value, sum;
    int left = -1, right = -1, height = 1;
  };

 public:
  DynamicFenwickTree() = default;
  explicit DynamicFenwickTree(Coord n) : n_(n) { assert(n >= 0); }
  DynamicFenwickTree(const DynamicFenwickTree&) = default;
  DynamicFenwickTree& operator=(const DynamicFenwickTree&) = default;
  DynamicFenwickTree(DynamicFenwickTree&& other)
      : n_(std::exchange(other.n_, 0)), root_(std::exchange(other.root_, -1)),
        nodes_(std::move(other.nodes_)) { other.nodes_.clear(); }
  DynamicFenwickTree& operator=(DynamicFenwickTree&& other) {
    if (this == &other) return *this;
    n_ = std::exchange(other.n_, 0);
    root_ = std::exchange(other.root_, -1);
    nodes_ = std::move(other.nodes_);
    other.nodes_.clear();
    return *this;
  }

  Coord size() const { return n_; }
  void add(Coord p, const T& value) {
    assert(0 <= p && p < n_);
    root_ = add_node(root_, p, value);
  }
  T pref(Coord r) const {
    assert(0 <= r && r <= n_);
    T answer = 0;
    int v = root_;
    while (v != -1) {
      const Node& node = nodes_[v];
      if (node.key < r) {
        answer += sum_of(node.left) + node.value;
        v = node.right;
      } else {
        v = node.left;
      }
    }
    return answer;
  }
  T sum(Coord l, Coord r) const {
    assert(0 <= l && l <= r && r <= n_);
    if (l == r) return 0;
    int v = root_;
    while (v != -1 && (nodes_[v].key < l || nodes_[v].key >= r))
      v = nodes_[v].key < l ? nodes_[v].right : nodes_[v].left;
    if (v == -1) return 0;
    T answer = nodes_[v].value;
    int left = nodes_[v].left, right = nodes_[v].right;
    while (left != -1) {
      const Node& node = nodes_[left];
      if (node.key >= l) {
        answer += node.value + sum_of(node.right);
        left = node.left;
      } else {
        left = node.right;
      }
    }
    while (right != -1) {
      const Node& node = nodes_[right];
      if (node.key < r) {
        answer += node.value + sum_of(node.left);
        right = node.right;
      } else {
        right = node.left;
      }
    }
    return answer;
  }
  T get(Coord p) const {
    assert(0 <= p && p < n_);
    int v = root_;
    while (v != -1) {
      const Node& node = nodes_[v];
      if (p < node.key) v = node.left;
      else if (node.key < p) v = node.right;
      else return node.value;
    }
    return 0;
  }
  Coord lower_bound(T weight) const {
    if (weight <= 0) return 0;
    int v = root_;
    while (v != -1) {
      const Node& node = nodes_[v];
      T left = sum_of(node.left);
      if (left >= weight) { v = node.left; continue; }
      weight -= left;
      if (node.value >= weight) return node.key;
      weight -= node.value;
      v = node.right;
    }
    return n_;
  }

 private:
  int height_of(int v) const { return v < 0 ? 0 : nodes_[v].height; }
  T sum_of(int v) const { return v < 0 ? 0 : nodes_[v].sum; }
  void pull(int v) {
    Node& node = nodes_[v];
    node.height = 1 + std::max(height_of(node.left), height_of(node.right));
    node.sum = sum_of(node.left) + node.value + sum_of(node.right);
  }
  int rotate(int v, bool right) {
    int u = right ? nodes_[v].left : nodes_[v].right;
    if (right) { nodes_[v].left = nodes_[u].right; nodes_[u].right = v; }
    else { nodes_[v].right = nodes_[u].left; nodes_[u].left = v; }
    pull(v); pull(u);
    return u;
  }
  int add_node(int v, Coord key, const T& value) {
    if (v == -1) {
      assert(nodes_.size() < static_cast<std::size_t>(std::numeric_limits<int>::max()));
      nodes_.push_back({key, value, value});
      return static_cast<int>(nodes_.size()) - 1;
    }
    // No reference into nodes_ survives recursion, which may reallocate it.
    if (key < nodes_[v].key) {
      int child = add_node(nodes_[v].left, key, value);
      nodes_[v].left = child;
    } else if (nodes_[v].key < key) {
      int child = add_node(nodes_[v].right, key, value);
      nodes_[v].right = child;
    } else {
      nodes_[v].value += value;
    }
    pull(v);
    const int balance = height_of(nodes_[v].left) - height_of(nodes_[v].right);
    if (balance > 1) {
      int child = nodes_[v].left;
      if (height_of(nodes_[child].left) < height_of(nodes_[child].right))
        nodes_[v].left = rotate(child, false);
      return rotate(v, true);
    }
    if (balance < -1) {
      int child = nodes_[v].right;
      if (height_of(nodes_[child].right) < height_of(nodes_[child].left))
        nodes_[v].right = rotate(child, true);
      return rotate(v, false);
    }
    return v;
  }
  Coord n_ = 0;
  int root_ = -1;
  std::vector<Node> nodes_;
};

}  // namespace blueberry
